# maps, the legacy-amd CWP table). Crawlers read them as links and fetched the
# bare directories, which 404 (Search Console "Not found", Nov 2026). Only the
# directory urls are blocked: the files inside stay crawlable, so rendering a
# page is unaffected.

Disallow: /*/dist/$
